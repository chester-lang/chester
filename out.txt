{
  record ParseResult [T] (result : T , remaining : List (Token));
  // // We need a helper to check if a token is a specific symbol
  def is_symbol_token (tok : Token , sym : String) : Bool = {
    string_eq (tok . text , sym);
    Unit
  };
  // // advance removes the first token
  def advance (tokens : List (Token)) : List (Token) = {
    let len = prim__list_length (tokens);
    Unit;
    if prim__int_eq (len , 0) then tokens else prim__list_make (len - 1 , \ i => prim__list_get (tokens , i + 1));
    Unit
  };
  // // We need to mutually recurse, so we define all functions
  // // Actually, Chester supports top-level mutual recursion
  def parse_tuple_helper (tokens : List (Token) , acc : List (CST)) : ParseResult (CST) = {
    if prim__int_eq (prim__list_length (tokens) , 0) then {
      ParseResult (CST . Error ("Unexpected EOF" , Span (0 , 0)) , tokens);
      Unit
    } else {
      let tok = prim__list_get (tokens , 0);
      Unit;
      match tok . kind {
        case Symbol (s) => {
          if string_eq (s , ")") then {
            ParseResult (CST . Tuple (acc , Span (0 , 0)) , advance (tokens));
            Unit
          } else {
            let p = parse_atom_sequence (tokens);
            Unit;
            match p . result {
              case Error (m , sp) => p;
              case _ => {
                if prim__int_eq (prim__list_length (p . remaining) , 0) then {
                  ParseResult (CST . Error ("Unexpected EOF in tuple" , Span (0 , 0)) , p . remaining);
                  Unit
                } else {
                  let next_tok = prim__list_get (p . remaining , 0);
                  Unit;
                  match next_tok . kind {
                    case Symbol (s2) => {
                      if string_eq (s2 , ",") then {
                        parse_tuple_helper (advance (p . remaining) , list_append (acc , p . result));
                        Unit
                      } else if string_eq (s2 , ")") then {
                        ParseResult (CST . Tuple (list_append (acc , p . result) , Span (0 , 0)) , advance (p . remaining));
                        Unit
                      } else {
                        ParseResult (CST . Error ("Expected ',' or ')'" , next_tok . span) , p . remaining);
                        Unit
                      };
                      Unit
                    };
                    case _ => ParseResult (CST . Error ("Expected ',' or ')'" , next_tok . span) , p . remaining);
                    Unit
                  };
                  Unit
                };
                Unit
              };
              Unit
            };
            Unit
          };
          Unit
        };
        case _ => {
          let p = parse_atom_sequence (tokens);
          Unit;
          match p . result {
            case Error (m , sp) => p;
            case _ => {
              if prim__int_eq (prim__list_length (p . remaining) , 0) then {
                ParseResult (CST . Error ("Unexpected EOF in tuple" , Span (0 , 0)) , p . remaining);
                Unit
              } else {
                let next_tok = prim__list_get (p . remaining , 0);
                Unit;
                match next_tok . kind {
                  case Symbol (s2) => {
                    if string_eq (s2 , ",") then {
                      parse_tuple_helper (advance (p . remaining) , list_append (acc , p . result));
                      Unit
                    } else if string_eq (s2 , ")") then {
                      ParseResult (CST . Tuple (list_append (acc , p . result) , Span (0 , 0)) , advance (p . remaining));
                      Unit
                    } else {
                      ParseResult (CST . Error ("Expected ',' or ')'" , next_tok . span) , p . remaining);
                      Unit
                    };
                    Unit
                  };
                  case _ => ParseResult (CST . Error ("Expected ',' or ')'" , next_tok . span) , p . remaining);
                  Unit
                };
                Unit
              };
              Unit
            };
            Unit
          };
          Unit
        };
        Unit
      };
      Unit
    };
    Unit
  };
  def parse_tuple (tokens : List (Token)) : ParseResult (CST) = {
    parse_tuple_helper (tokens , list_empty [CST] ());
    Unit
  };
  def parse_block_helper (tokens : List (Token) , acc : List (CST)) : ParseResult (CST) = {
    if prim__int_eq (prim__list_length (tokens) , 0) then {
      ParseResult (CST . Error ("Unexpected EOF" , Span (0 , 0)) , tokens);
      Unit
    } else {
      let tok = prim__list_get (tokens , 0);
      Unit;
      match tok . kind {
        case Symbol (s) => {
          if string_eq (s , "}") then {
            ParseResult (CST . Block (acc , list_empty [CST] () , Span (0 , 0)) , advance (tokens));
            Unit
          } else {
            let p = parse_atom_sequence (tokens);
            Unit;
            match p . result {
              case Error (m , sp) => p;
              case _ => {
                if prim__int_eq (prim__list_length (p . remaining) , 0) then {
                  ParseResult (CST . Error ("Unexpected EOF in block" , Span (0 , 0)) , p . remaining);
                  Unit
                } else {
                  let next_tok = prim__list_get (p . remaining , 0);
                  Unit;
                  match next_tok . kind {
                    case Symbol (s2) => {
                      if string_eq (s2 , ";") then {
                        parse_block_helper (advance (p . remaining) , list_append (acc , p . result));
                        Unit
                      } else if string_eq (s2 , "}") then {
                        // // tail expression!
                        let tail = list_empty [CST] ();
                        Unit;
                        ParseResult (CST . Block (acc , list_append (tail , p . result) , Span (0 , 0)) , advance (p . remaining));
                        Unit
                      } else {
                        ParseResult (CST . Error ("Expected ';' or '}'" , next_tok . span) , p . remaining);
                        Unit
                      };
                      Unit
                    };
                    case _ => ParseResult (CST . Error ("Expected ';' or '}'" , next_tok . span) , p . remaining);
                    Unit
                  };
                  Unit
                };
                Unit
              };
              Unit
            };
            Unit
          };
          Unit
        };
        case _ => {
          let p = parse_atom_sequence (tokens);
          Unit;
          match p . result {
            case Error (m , sp) => p;
            case _ => {
              if prim__int_eq (prim__list_length (p . remaining) , 0) then {
                ParseResult (CST . Error ("Unexpected EOF in block" , Span (0 , 0)) , p . remaining);
                Unit
              } else {
                let next_tok = prim__list_get (p . remaining , 0);
                Unit;
                match next_tok . kind {
                  case Symbol (s2) => {
                    if string_eq (s2 , ";") then {
                      parse_block_helper (advance (p . remaining) , list_append (acc , p . result));
                      Unit
                    } else if string_eq (s2 , "}") then {
                      // // tail expression!
                      let tail = list_empty [CST] ();
                      Unit;
                      ParseResult (CST . Block (acc , list_append (tail , p . result) , Span (0 , 0)) , advance (p . remaining));
                      Unit
                    } else {
                      ParseResult (CST . Error ("Expected ';' or '}'" , next_tok . span) , p . remaining);
                      Unit
                    };
                    Unit
                  };
                  case _ => ParseResult (CST . Error ("Expected ';' or '}'" , next_tok . span) , p . remaining);
                  Unit
                };
                Unit
              };
              Unit
            };
            Unit
          };
          Unit
        };
        Unit
      };
      Unit
    };
    Unit
  };
  def parse_block (tokens : List (Token)) : ParseResult (CST) = {
    parse_block_helper (tokens , list_empty [CST] ());
    Unit
  };
  def is_terminator (tok : Token) : Bool = {
    match tok . kind {
      case EOF => true;
      case Symbol (s) => {
        let is_rparen = string_eq (s , ")");
        Unit;
        let is_rbrace = string_eq (s , "}");
        Unit;
        let is_rbracket = string_eq (s , "]");
        Unit;
        let is_semi = string_eq (s , ";");
        Unit;
        let is_comma = string_eq (s , ",");
        Unit;
        bool_or (is_rparen , bool_or (is_rbrace , bool_or (is_rbracket , bool_or (is_semi , is_comma))));
        Unit
      };
      case _ => false;
      Unit
    };
    Unit
  };
  def parse_atom_sequence_helper (tokens : List (Token) , acc : List (CST)) : ParseResult (CST) = {
    if prim__int_eq (prim__list_length (tokens) , 0) then {
      ParseResult (CST . Error ("Unexpected EOF" , Span (0 , 0)) , tokens);
      Unit
    } else {
      let tok = prim__list_get (tokens , 0);
      Unit;
      if is_terminator (tok) then {
        let len = prim__list_length (acc);
        Unit;
        if prim__int_eq (len , 1) then ParseResult (prim__list_get (acc , 0) , tokens) else ParseResult (CST . SeqOf (acc , Span (0 , 0)) , tokens);
        Unit
      } else {
        let p = parse_atom (tokens);
        Unit;
        match p . result {
          case Error (m , sp) => p;
          case _ => parse_atom_sequence_helper (p . remaining , list_append (acc , p . result));
          Unit
        };
        Unit
      };
      Unit
    };
    Unit
  };
  def parse_atom_sequence (tokens : List (Token)) : ParseResult (CST) = {
    parse_atom_sequence_helper (tokens , list_empty [CST] ());
    Unit
  };
  def parse_atom (tokens : List (Token)) : ParseResult (CST) = {
    let len = prim__list_length (tokens);
    Unit;
    if prim__int_eq (len , 0) then {
      ParseResult (CST . Error ("Unexpected EOF" , Span (0 , 0)) , tokens);
      Unit
    } else {
      let tok = prim__list_get (tokens , 0);
      Unit;
      let rest = advance (tokens);
      Unit;
      match tok . kind {
        case Identifier => ParseResult (CST . Symbol (tok . text , tok . span) , rest);
        case IntegerLit => ParseResult (CST . IntegerLiteral (tok . text , tok . span) , rest);
        case StringLit => ParseResult (CST . StringLiteral (tok . text , tok . span) , rest);
        case EOF => ParseResult (CST . Error ("Unexpected EOF" , tok . span) , rest);
        case Symbol (s) => {
          if string_eq (s , "(") then parse_tuple (rest) else if string_eq (s , "{") then parse_block (rest) else ParseResult (CST . Symbol (tok . text , tok . span) , rest);
          Unit
        };
        case Whitespace => ParseResult (CST . Error ("Unexpected whitespace" , tok . span) , rest) // // Whitespace should be filtered;
        Unit
      };
      Unit
    };
    Unit
  };
  def filter_whitespace (tokens : List (Token)) : List (Token) = {
    list_filter (tokens , \ tok => match tok . kind {
      case Whitespace => false;
      case _ => true;
      Unit
    });
    Unit
  };
  def parse (tokens : List (Token)) : ParseResult (CST) = {
    let filtered = filter_whitespace (tokens);
    Unit;
    parse_atom_sequence (filtered);
    Unit
  };
  def main_parser () : Any = {
    let tokens = lex ("let x = 5;");
    Unit;
    let p = parse (tokens);
    Unit;
    // // Just return p to check if it compiles!
    p;
    Unit
  };
  main_parser ();
  Unit
